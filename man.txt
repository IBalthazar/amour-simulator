const state = {
  surveyAnswers: {},
  scenarioChoices: [],
  surveyIndex: 0,
  scenarioIndex: 0,
  selectedScenario: null
};

function showScreen(name) {
  screens.forEach((screen) => {
    screen.hidden = screen.dataset.screen !== name;
  });
  window.scrollTo({ top: 0, behavior: 'smooth' });
  mainNav.classList.remove('open');
  menuToggle.setAttribute('aria-expanded', 'false');
}

function renderSurvey() {
  const question = surveyQuestions[state.surveyIndex];
  const form = document.querySelector('#survey-form');
  document.querySelector('#survey-step').textContent = `QUESTION ${state.surveyIndex + 1} OF ${surveyQuestions.length}`;
  form.innerHTML = `<fieldset><legend>${question.prompt}</legend>${question.options.map((option) => `
    <label class="answer-option ${state.surveyAnswers[question.id] === option.value ? 'selected' : ''}">
      <input type="radio" name="${question.id}" value="${option.value}" ${state.surveyAnswers[question.id] === option.value ? 'checked' : ''}>
      <span>${option.label}</span><span class="answer-check" aria-hidden="true">✓</span>
    </label>`).join('')}</fieldset>`;
  form.querySelectorAll('input').forEach((input) => {
    input.addEventListener('change', () => {
      form.querySelectorAll('.answer-option').forEach((option) => option.classList.remove('selected'));
      input.closest('.answer-option').classList.add('selected');
      document.querySelector('#survey-next').disabled = false;
    });
  });
  document.querySelector('#survey-next').disabled = !state.surveyAnswers[question.id];
}

function startSurvey() {
  state.surveyAnswers = {};
  state.scenarioChoices = [];
  state.surveyIndex = 0;
  state.scenarioIndex = 0;
  renderSurvey();
  showScreen('survey');
}

function renderScenario() {
  const scenario = scenarios[state.scenarioIndex];
  document.querySelector('#scenario-step').textContent = `SCENARIO ${state.scenarioIndex + 1} OF ${scenarios.length}`;
  document.querySelector('#scenario-progress').style.width = `${((state.scenarioIndex + 1) / scenarios.length) * 100}%`;
  document.querySelector('#scenario-title').textContent = scenario.title;
  document.querySelector('#scenario-context').textContent = scenario.context;
  document.querySelector('#scenario-options').innerHTML = scenario.options.map((option, index) => `
    <button class="scenario-option ${state.selectedScenario === index ? 'selected' : ''}" type="button" data-option="${index}">
      <span class="option-icon ${index === 1 ? 'orange' : ''}">${option.icon}</span>
      <span>${option.label}</span><span class="select-indicator ${state.selectedScenario === index ? '' : 'empty'}">${state.selectedScenario === index ? '✓' : ''}</span>
    </button>`).join('');
  const nextButton = document.querySelector('#scenario-next');
  nextButton.disabled = state.selectedScenario === null;
  nextButton.innerHTML = `${state.scenarioIndex === scenarios.length - 1 ? 'See my results' : 'Choose a response'} <img src="assets/arrow-right.svg" alt="">`;
  document.querySelectorAll('[data-option]').forEach((option) => {
    option.addEventListener('click', () => {
      state.selectedScenario = Number(option.dataset.option);
      renderScenario();
    });
  });
}

function startScenarios() {
  state.scenarioIndex = 0;
  state.scenarioChoices = [];
  state.selectedScenario = null;
  renderScenario();
  showScreen('scenarios');
}

function getDominantSignal() {
  const scores = {};
  surveyQuestions.forEach((question) => {
    const answer = question.options.find((option) => option.value === state.surveyAnswers[question.id]);
    if (answer) scores[answer.signal] = (scores[answer.signal] || 0) + 2;
  });
  state.scenarioChoices.forEach((choice) => {
    scores[choice.metric] = (scores[choice.metric] || 0) + 1;
  });
  return Object.entries(scores).sort((a, b) => b[1] - a[1])[0]?.[0] || 'openness';
}

function renderResults() {
  const signal = getDominantSignal();
  const archetype = archetypes[signal];
  document.querySelector('#archetype-name').textContent = archetype.name;
  document.querySelector('#archetype-description').textContent = archetype.description;
  document.querySelector('#archetype-traits').innerHTML = archetype.traits.map((trait) => `<span>${trait}</span>`).join('');
  document.querySelector('#results-intro').textContent = `You brought ${state.scenarioChoices.length} choices and ${Object.keys(state.surveyAnswers).length} starting beliefs to the table. Here is the pattern we noticed.`;
  document.querySelector('#comparison-list').innerHTML = surveyQuestions.map((question) => {
    const answer = question.options.find((option) => option.value === state.surveyAnswers[question.id]);
    const choice = state.scenarioChoices.find((item) => item.metric === answer?.signal);
    return `<div class="comparison-item"><span>${question.prompt}</span><strong>${answer?.label || 'Not answered'}</strong><small>${choice ? `Your choices echoed this: ${choice.tag}.` : 'Your scenarios added a new angle to this belief.'}</small></div>`;
  }).join('');
  showScreen('results');
}

document.querySelector('#survey-form').addEventListener('submit', (event) => {
  event.preventDefault();
  const question = surveyQuestions[state.surveyIndex];
  const selected = new FormData(event.currentTarget).get(question.id);
  if (!selected) return;
  state.surveyAnswers[question.id] = selected;
  if (state.surveyIndex < surveyQuestions.length - 1) {
    state.surveyIndex += 1;
    renderSurvey();
  } else {
    startScenarios();
  }
});

document.querySelector('#scenario-next').addEventListener('click', () => {
  if (state.selectedScenario === null) return;
  const selected = scenarios[state.scenarioIndex].options[state.selectedScenario];
  state.scenarioChoices.push(selected);
  if (state.scenarioIndex < scenarios.length - 1) {
    state.scenarioIndex += 1;
    state.selectedScenario = null;
    renderScenario();
  } else {
    renderResults();
  }
});

document.querySelector('#play-again').addEventListener('click', startSurvey);
document.querySelector('#back-to-home').addEventListener('click', () => showScreen('welcome'));

  control.addEventListener('click', () => {
    if (control.dataset.route === 'survey') startSurvey();
    else showScreen(control.dataset.route);
  });
});
